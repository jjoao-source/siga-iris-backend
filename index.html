// LISTA DE FOTOS PARA O BACKGROUND DA TELA DE LOGIN const imagensBackground = [ "logo.png", "logo1.png", "logo2.png", "logo3.png", "logo4.png", "logo5.png", "logo6.png", "logo7.png", "logo8.png", "logo9.png", "logo10.png", "logo11.png", "logo12.png" ]; let indiceFundoAtual = 0; function aplicarBackground(url) { const containerLogin = document.getElementById("tela-login"); if (containerLogin) { containerLogin.style.backgroundImage = `linear-gradient(rgba(15, 23, 42, 0.75), rgba(15, 23, 42, 0.75)), url('${url}')`; } } function trocarFundo() { const nomeImagem = imagensBackground[indiceFundoAtual]; const imgLocal = `./static/${nomeImagem}`; const imgRemota = `${API_URL}/static/${nomeImagem}`; // Tenta carregar imagem local primeiro const img = new Image(); img.src = imgLocal; img.onload = () => aplicarBackground(imgLocal); img.onerror = () => aplicarBackground(imgRemota); indiceFundoAtual = (indiceFundoAtual + 1) % imagensBackground.length; } // Inicializa a troca de imagens trocarFundo(); setInterval(trocarFundo, 5000); // Carregamento garantido do Logo const imgLogoOficial = document.getElementById("img-logo-oficial"); const imgLogoHeader = document.getElementById("img-logo-header"); if (imgLogoOficial && imgLogoHeader) { const logoLocal = "./static/logo_universidade.png"; const logoRemoto = `${API_URL}/static/logo_universidade.png`; const logoTest = new Image(); logoTest.src = logoLocal; logoTest.onload = () => { imgLogoOficial.src = logoLocal; imgLogoHeader.src = logoLocal; }; logoTest.onerror = () => { imgLogoOficial.src = logoRemoto; imgLogoHeader.src = logoRemoto; }; }
